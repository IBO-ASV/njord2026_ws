cmake_minimum_required(VERSION 3.8)
project(ground_station_heartbeat)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
add_executable(ground_station_heartbeat_node src/ground_station_heartbeat.cpp)
ament_target_dependencies(ground_station_heartbeat_node rclcpp std_msgs)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
endif()
install(TARGETS ground_station_heartbeat_node ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
ament_package()
