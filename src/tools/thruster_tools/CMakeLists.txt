cmake_minimum_required(VERSION 3.8)
project(thruster_tools)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
add_executable(thruster_id_test src/thruster_id_test.cpp)
ament_target_dependencies(thruster_id_test rclcpp std_msgs)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
endif()
install(TARGETS thruster_id_test ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
ament_package()
