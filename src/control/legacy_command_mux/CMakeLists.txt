cmake_minimum_required(VERSION 3.8)
project(legacy_command_mux)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
add_executable(legacy_command_mux_node src/command_arbiter.cpp)
ament_target_dependencies(legacy_command_mux_node rclcpp geometry_msgs std_msgs)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
endif()
install(TARGETS legacy_command_mux_node ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
ament_package()
