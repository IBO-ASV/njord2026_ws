cmake_minimum_required(VERSION 3.8)
project(joystick_control)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
add_library(joystick_input src/joy_converter.cpp)
target_include_directories(joystick_input PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(joystick_input rclcpp sensor_msgs geometry_msgs std_msgs)
add_executable(joy_converter_node src/main.cpp)
target_include_directories(joy_converter_node PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(joy_converter_node rclcpp sensor_msgs geometry_msgs std_msgs)
target_link_libraries(joy_converter_node joystick_input)
add_executable(sbus_joy_converter_node src/sbus_joy_converter.cpp)
target_include_directories(sbus_joy_converter_node PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(sbus_joy_converter_node rclcpp sensor_msgs geometry_msgs std_msgs)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_joy_converter test/test_joy_converter.cpp)
  target_include_directories(test_joy_converter PRIVATE include)
  ament_target_dependencies(test_joy_converter geometry_msgs rclcpp sensor_msgs std_msgs)
  target_link_libraries(test_joy_converter joystick_input)
  ament_add_gtest(test_sbus_joy_converter test/test_sbus_joy_converter.cpp src/sbus_joy_converter.cpp)
  target_include_directories(test_sbus_joy_converter PRIVATE include)
  ament_target_dependencies(test_sbus_joy_converter geometry_msgs rclcpp sensor_msgs std_msgs)
  target_compile_definitions(test_sbus_joy_converter PRIVATE main=sbus_joy_converter_main)
endif()
install(TARGETS joystick_input joy_converter_node sbus_joy_converter_node ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
ament_export_include_directories(include)
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
ament_package()
